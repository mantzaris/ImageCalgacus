% Conference page placement, verified against the supplied ICBDAIA 2027 Word template.
% This file does NOT change the Springer 122 x 193 mm text block, font sizes,
% paragraph/heading/caption spacing, or bibliography formatting.
% Load after \documentclass{llncs} and the official font packages.
\usepackage[
  a4paper,
  textwidth=122mm,
  textheight=193mm,
  hcentering,
  vcentering,
  headheight=12pt,
  headsep=16.45276pt
]{geometry}
\usepackage{fancyhdr}
\usepackage{etoolbox}

% The Word template has a distinct first-page header. Its conference name is 8 pt
% bold; later pages have 9 pt outer page numbers, with no running title/name.
% Header box starts 42 mm from the top; body starts 52 mm from the top.
\fancypagestyle{icbdaiafirst}{%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
  \fancyhead[C]{\fontsize{8}{10}\selectfont\bfseries
    The 6\textsuperscript{th} International Conference on Big Data and
    Artificial Intelligence Applications (ICBDAIA'27)}%
}
\fancypagestyle{icbdaiapages}{%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
  \fancyhead[LE,RO]{\small\thepage}%
}
\AtBeginDocument{\pagestyle{icbdaiapages}}
% The unmodified Springer \maketitle explicitly selects an empty page style.
% Apply only the conference's first-page header after that command finishes.
\apptocmd{\maketitle}{\thispagestyle{icbdaiafirst}}{}{
  \PackageError{icbdaia}{Cannot attach conference first-page header}{}
}
